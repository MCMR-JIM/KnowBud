import asyncio
import edge_tts
import argparse

async def tts_run(text, output="edge_tts_demo.mp3"):
    communicate = edge_tts.Communicate(text, "zh-CN-XiaoxiaoNeural")
    await communicate.save(output)
    print(f"保存成功：{output}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--text", default="你好，这是文字转语音测试")
    parser.add_argument("--output", default="edge_tts_demo.mp3")
    args = parser.parse_args()
    asyncio.run(tts_run(args.text, args.output))
